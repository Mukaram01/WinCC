// UpdateStartTime.c — writes "HH:MM" and "DD/MM/YYYY"
#include "apdefap.h"

void UpdateStartTime(const char* timeTag, const char* dateTag)
{
    time_t now;
    struct tm* tm_info;
    char formattedTime[6];
    char formattedDate[11];

    now = time(NULL);
    tm_info = localtime(&now);
    if (!tm_info) return;

    strftime(formattedTime, sizeof(formattedTime), "%H:%M", tm_info);
    strftime(formattedDate, sizeof(formattedDate), "%d/%m/%Y", tm_info);

    if (timeTag && timeTag[0] != '\0') SetTagChar(timeTag, formattedTime);
    if (dateTag && dateTag[0] != '\0') SetTagChar(dateTag, formattedDate);
}
