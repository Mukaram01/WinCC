// CalculateEstimatedEndTime.c — ETA (seconds) from ProductionEvents history
#include "apdefap.h"

#define MIN_REQUIRED_SAMPLES 3
#define DEFAULT_ETA_SECONDS  3600
#define MAX_RUNTIME_SAMPLES  2048
#define MAX_ROW_TOTAL_MINUTES  (24 * 60)
#define PLANNED_WINDOW_MINUTES (24 * 60)
#define WINDOW_TOLERANCE_MINUTES 0
#define QUARANTINE_ARCHIVE_NAME "ProductionEvents_Quarantine"

#define QUARANTINE_F_ROW_ID   1
#define QUARANTINE_F_PAYLOAD  2
#define QUARANTINE_F_REASON   3

/* ProductionEvents field indices (1-based) */
#define F_OPERATION      4
#define F_PART           5
#define F_RUNTIME_HHMM   6
#define F_DOWNTIME_HHMM  7

static int parseRuntimeSeconds(const char* runtime,
                               long rowId,
                               const char* fieldName,
                               double* seconds,
                               char* normalizedHHMM,
                               unsigned int normalizedSize,
                               char* rejectReason,
                               unsigned int rejectReasonSize);
static void copyTrim(char* dest, unsigned int destSize, const char* src);
static unsigned int textLength(const char* text);
static int textCompare(const char* a, const char* b);
static int isSpaceChar(char c);
static const char* skipSpace(const char* text);
static int parseNonNegativeLong(const char* text, const char** endOut, long* valueOut);
static void minutesToHHMM(long totalMinutes, char* out, unsigned int outSize);
static int quarantineRow(const char* quarantineArchiveName, long rowId, const char* payload, const char* reason);
static void sortDoubles(double* values, int count);

void CalculateEstimatedEndTime(const char* archiveName,
                               const char* operationTag,
                               const char* partTag,
                               const char* resultTag)
{
    UAHCONNECT hConnect = 0;
    UAHARCHIVE hArchive = 0;
    double runTimes[MAX_RUNTIME_SAMPLES];
    int storedSamples = 0;
    int nextIndex = 0;
    double median = DEFAULT_ETA_SECONDS;
    int archiveOpened = 0;
    double sumSamples = 0.0;
    long sourceRowId = 0;

    const char* operationNumberSrc = GetTagChar(operationTag);
    const char* partNumberSrc      = GetTagChar(partTag);
    char operationNumber[64];
    char partNumber[64];

    if (!archiveName || !*archiveName || !operationTag || !*operationTag || !partTag || !*partTag) {
        SetTagDouble(resultTag, DEFAULT_ETA_SECONDS);
        return;
    }

    copyTrim(operationNumber, sizeof(operationNumber), operationNumberSrc ? operationNumberSrc : "");
    copyTrim(partNumber, sizeof(partNumber), partNumberSrc ? partNumberSrc : "");

    if (!operationNumber[0] || !partNumber[0]) {
        SetTagDouble(resultTag, DEFAULT_ETA_SECONDS);
        return;
    }

    if (!uaConnect(&hConnect)) goto cleanup;
    if (!uaQueryArchiveByName(hConnect, archiveName, &hArchive)) goto cleanup;
    if (!uaArchiveOpen(hArchive)) goto cleanup;
    archiveOpened = 1;

    if (uaArchiveMoveFirst(hArchive)) {
        do {
            char bufOp[64] = "";
            char bufPart[64] = "";
            char bufRuntime[32] = "";
            char bufDowntime[32] = "";
            char normalizedRuntime[16] = "";
            char normalizedDowntime[16] = "";
            char rejectReason[160] = "";
            char sourcePayload[256] = "";
            double runtimeSeconds = 0.0;
            double downtimeSeconds = 0.0;
            double totalSeconds = 0.0;
            int runtimeOk = 0;
            int downtimeOk = 0;

            ++sourceRowId;

            uaArchiveGetFieldValueString(hArchive, F_OPERATION, bufOp, sizeof(bufOp));
            uaArchiveGetFieldValueString(hArchive, F_PART,      bufPart, sizeof(bufPart));

            if (textCompare(bufOp, operationNumber) == 0 && textCompare(bufPart, partNumber) == 0) {
                uaArchiveGetFieldValueString(hArchive, F_RUNTIME_HHMM,  bufRuntime,  sizeof(bufRuntime));
                uaArchiveGetFieldValueString(hArchive, F_DOWNTIME_HHMM, bufDowntime, sizeof(bufDowntime));

                runtimeOk = parseRuntimeSeconds(bufRuntime,
                                                sourceRowId,
                                                "RuntimeHHMM",
                                                &runtimeSeconds,
                                                normalizedRuntime,
                                                sizeof(normalizedRuntime),
                                                rejectReason,
                                                sizeof(rejectReason));
                downtimeOk = parseRuntimeSeconds(bufDowntime,
                                                 sourceRowId,
                                                 "DowntimeHHMM",
                                                 &downtimeSeconds,
                                                 normalizedDowntime,
                                                 sizeof(normalizedDowntime),
                                                 rejectReason,
                                                 sizeof(rejectReason));

                if (!runtimeOk || !downtimeOk) {
                    sprintf(sourcePayload,
                            "rowId=%ld,operation=%s,part=%s,runtime=%s,downtime=%s,normRuntime=%s,normDowntime=%s",
                            sourceRowId, bufOp, bufPart, bufRuntime, bufDowntime, normalizedRuntime, normalizedDowntime);
                    quarantineRow(QUARANTINE_ARCHIVE_NAME, sourceRowId, sourcePayload, rejectReason);
                    printf("CalculateEstimatedEndTime: quarantined rowId=%ld reason=%s payload=%s\n",
                           sourceRowId, rejectReason, sourcePayload);
                    continue;
                }

                totalSeconds = runtimeSeconds + downtimeSeconds;
                if (totalSeconds > 0.0) {
                    if (storedSamples < MAX_RUNTIME_SAMPLES) {
                        runTimes[storedSamples++] = totalSeconds;
                        sumSamples += totalSeconds;
                        nextIndex = storedSamples;
                    } else if (MAX_RUNTIME_SAMPLES > 0) {
                        if (nextIndex >= MAX_RUNTIME_SAMPLES) {
                            nextIndex = 0;
                        }
                        sumSamples -= runTimes[nextIndex];
                        runTimes[nextIndex++] = totalSeconds;
                        sumSamples += totalSeconds;
                    }

                    if (nextIndex >= MAX_RUNTIME_SAMPLES) {
                        nextIndex = 0;
                    }
                }
            }
        } while (uaArchiveMoveNext(hArchive));
    }

    if (storedSamples >= MIN_REQUIRED_SAMPLES) {
        sortDoubles(runTimes, storedSamples);
        if (storedSamples & 1) {
            median = runTimes[storedSamples / 2];
        } else {
            median = 0.5 * (runTimes[storedSamples / 2 - 1] + runTimes[storedSamples / 2]);
        }
    } else if (storedSamples > 0) {
        median = sumSamples / (double)storedSamples;
    } else {
        median = DEFAULT_ETA_SECONDS;
    }

cleanup:
    if (hArchive) {
        if (archiveOpened) uaArchiveClose(hArchive);
        uaReleaseArchive(hArchive);
    }
    if (hConnect) {
        uaDisconnect(hConnect);
    }
    SetTagDouble(resultTag, median);
}

static void sortDoubles(double* values, int count)
{
    int i;
    for (i = 1; i < count; ++i) {
        double current = values[i];
        int j = i - 1;
        while (j >= 0 && values[j] > current) {
            values[j + 1] = values[j];
            --j;
        }
        values[j + 1] = current;
    }
}

static int parseRuntimeSeconds(const char* runtime,
                               long rowId,
                               const char* fieldName,
                               double* seconds,
                               char* normalizedHHMM,
                               unsigned int normalizedSize,
                               char* rejectReason,
                               unsigned int rejectReasonSize)
{
    const char* p;
    const char* colon;
    long hours = 0;
    long minutes = 0;
    long totalMinutes = 0;
    long maxPlannedWithTolerance = PLANNED_WINDOW_MINUTES + WINDOW_TOLERANCE_MINUTES;

    if (!runtime || !*runtime || !seconds || !fieldName || !rejectReason || rejectReasonSize == 0) {
        if (rejectReason && rejectReasonSize > 0) {
            sprintf(rejectReason, "rowId=%ld malformed-empty", rowId);
        }
        return 0;
    }

    p = skipSpace(runtime);
    colon = p;
    while (*colon && *colon != ':') {
        ++colon;
    }

    if (*colon != ':') {
        sprintf(rejectReason, "rowId=%ld malformed-no-colon field=%s value=%s", rowId, fieldName, runtime);
        return 0;
    }

    if (!parseNonNegativeLong(p, &colon, &hours)) {
        sprintf(rejectReason, "rowId=%ld malformed-hour field=%s value=%s", rowId, fieldName, runtime);
        return 0;
    }

    colon = skipSpace(colon);
    if (*colon != ':') {
        sprintf(rejectReason, "rowId=%ld malformed-hour-separator field=%s value=%s", rowId, fieldName, runtime);
        return 0;
    }

    p = skipSpace(colon + 1);
    if (!parseNonNegativeLong(p, &colon, &minutes)) {
        sprintf(rejectReason, "rowId=%ld malformed-minute field=%s value=%s", rowId, fieldName, runtime);
        return 0;
    }

    colon = skipSpace(colon);
    if (*colon != '\0' || minutes < 0 || minutes >= 60 || hours < 0) {
        sprintf(rejectReason, "rowId=%ld malformed-minute-range field=%s value=%s", rowId, fieldName, runtime);
        return 0;
    }

    totalMinutes = hours * 60 + minutes;
    if (totalMinutes > MAX_ROW_TOTAL_MINUTES || totalMinutes > maxPlannedWithTolerance) {
        sprintf(rejectReason,
                "rowId=%ld threshold-exceeded field=%s totalMinutes=%ld maxPerRow=%d maxWindow=%ld",
                rowId, fieldName, totalMinutes, MAX_ROW_TOTAL_MINUTES, maxPlannedWithTolerance);
        return 0;
    }

    minutesToHHMM(totalMinutes, normalizedHHMM, normalizedSize);
    *seconds = (double)totalMinutes * 60.0;
    return 1;
}

static void minutesToHHMM(long totalMinutes, char* out, unsigned int outSize)
{
    long hh;
    long mm;
    if (!out || outSize < 6 || totalMinutes < 0) {
        return;
    }

    hh = totalMinutes / 60;
    mm = totalMinutes % 60;
    sprintf(out, "%02ld:%02ld", hh, mm);
}

static int quarantineRow(const char* quarantineArchiveName, long rowId, const char* payload, const char* reason)
{
    UAHCONNECT hCon = 0;
    UAHARCHIVE hArc = 0;
    int ok = 0;
    char rowIdText[32] = "";

    if (!quarantineArchiveName || !*quarantineArchiveName || !payload || !reason) {
        return 0;
    }

    sprintf(rowIdText, "%ld", rowId);

    if (!uaConnect(&hCon)) goto cleanup;
    if (!uaQueryArchiveByName(hCon, quarantineArchiveName, &hArc)) goto cleanup;
    if (!uaArchiveOpen(hArc)) goto cleanup;

    uaArchiveSetFieldValueString(hArc, QUARANTINE_F_ROW_ID,  rowIdText);
    uaArchiveSetFieldValueString(hArc, QUARANTINE_F_PAYLOAD, payload);
    uaArchiveSetFieldValueString(hArc, QUARANTINE_F_REASON,  reason);
    ok = uaArchiveInsert(hArc);

cleanup:
    if (hArc) {
        uaArchiveClose(hArc);
        uaReleaseArchive(hArc);
    }
    if (hCon) {
        uaDisconnect(hCon);
    }
    return ok;
}

static int parseNonNegativeLong(const char* text, const char** endOut, long* valueOut)
{
    const char* p = text;
    long value = 0;
    int digits = 0;

    if (!text || !valueOut) {
        return 0;
    }

    while (*p >= '0' && *p <= '9') {
        long nextValue = value * 10 + (*p - '0');
        if (nextValue < value) {
            return 0;
        }
        value = nextValue;
        ++p;
        ++digits;
    }

    if (digits == 0) {
        return 0;
    }

    if (endOut) {
        *endOut = p;
    }

    *valueOut = value;
    return 1;
}

static const char* skipSpace(const char* text)
{
    const char* p = text;
    while (p && *p && isSpaceChar(*p)) {
        ++p;
    }
    return p ? p : text;
}

static int isSpaceChar(char c)
{
    return c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\f' || c == '\v';
}

static unsigned int textLength(const char* text)
{
    unsigned int len = 0;
    if (!text) {
        return 0;
    }

    while (text[len] != '\0') {
        ++len;
    }

    return len;
}

static int textCompare(const char* a, const char* b)
{
    unsigned char ca;
    unsigned char cb;

    if (!a && !b) return 0;
    if (!a) return -1;
    if (!b) return 1;

    while (*a || *b) {
        ca = (unsigned char)*a;
        cb = (unsigned char)*b;
        if (ca != cb) {
            return (ca < cb) ? -1 : 1;
        }
        if (*a == '\0') {
            break;
        }
        ++a;
        ++b;
    }

    return 0;
}

static void copyTrim(char* dest, unsigned int destSize, const char* src)
{
    const char* start;
    const char* end;
    unsigned int len;
    unsigned int i;

    if (!dest || destSize == 0) {
        return;
    }

    dest[0] = '\0';
    if (!src) {
        return;
    }

    start = src;
    while (*start && isSpaceChar(*start)) {
        ++start;
    }

    end = start + textLength(start);
    while (end > start && isSpaceChar(*(end - 1))) {
        --end;
    }

    len = (unsigned int)(end - start);
    if (len >= destSize) {
        len = destSize - 1;
    }

    for (i = 0; i < len; ++i) {
        dest[i] = start[i];
    }
    dest[len] = '\0';
}
