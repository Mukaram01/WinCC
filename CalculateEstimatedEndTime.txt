// CalculateEstimatedEndTime.c — same signature; median seconds
#include "apdefap.h"

#define MIN_REQUIRED_SAMPLES 3
#define DEFAULT_ETA_SECONDS  3600
#define MAX_RUNTIME_SAMPLES  2048

static int parseRuntimeSeconds(const char* runtime, double* seconds);
static void copyTrim(char* dest, unsigned int destSize, const char* src);
static unsigned int textLength(const char* text);
static int textCompare(const char* a, const char* b);
static int isSpaceChar(char c);
static const char* skipSpace(const char* text);
static int parseNonNegativeLong(const char* text, const char** endOut, long* valueOut);
static void sortDoubles(double* values, int count);

void CalculateEstimatedEndTime(const char* archiveName,
                               const char* operationTag,
                               const char* partTag,
                               const char* resultTag)
{
    UAHCONNECT hConnect = 0;
    UAHARCHIVE hArchive = 0;
    double runTimes[MAX_RUNTIME_SAMPLES];
    int storedSamples = 0;
    int nextIndex = 0;
    double median = DEFAULT_ETA_SECONDS;
    int archiveOpened = 0;
    double sumSamples = 0.0;

    const char* operationNumberSrc = GetTagChar(operationTag);
    const char* partNumberSrc      = GetTagChar(partTag);
    char operationNumber[64];
    char partNumber[64];

    if (!archiveName || !*archiveName || !operationTag || !*operationTag || !partTag || !*partTag) {
        SetTagDouble(resultTag, DEFAULT_ETA_SECONDS);
        return;
    }

    copyTrim(operationNumber, sizeof(operationNumber), operationNumberSrc ? operationNumberSrc : "");
    copyTrim(partNumber, sizeof(partNumber), partNumberSrc ? partNumberSrc : "");

    if (!operationNumber[0] || !partNumber[0]) {
        SetTagDouble(resultTag, DEFAULT_ETA_SECONDS);
        return;
    }

    if (!uaConnect(&hConnect)) goto cleanup;
    if (!uaQueryArchiveByName(hConnect, archiveName, &hArchive)) goto cleanup;
    if (!uaArchiveOpen(hArchive)) goto cleanup;
    archiveOpened = 1;

    if (uaArchiveMoveFirst(hArchive)) {
        do {
            char bufOp[64] = "";
            char bufPart[64] = "";
            char bufRuntime[32] = "";
            double seconds = 0.0;

            uaArchiveGetFieldValueString(hArchive, 4, bufOp, sizeof(bufOp));
            uaArchiveGetFieldValueString(hArchive, 5, bufPart, sizeof(bufPart));

            if (textCompare(bufOp, operationNumber) == 0 && textCompare(bufPart, partNumber) == 0) {
                uaArchiveGetFieldValueString(hArchive, 6, bufRuntime, sizeof(bufRuntime));

                if (parseRuntimeSeconds(bufRuntime, &seconds) && seconds > 0.0) {
                    if (storedSamples < MAX_RUNTIME_SAMPLES) {
                        runTimes[storedSamples++] = seconds;
                        sumSamples += seconds;
                        nextIndex = storedSamples;
                    } else if (MAX_RUNTIME_SAMPLES > 0) {
                        if (nextIndex >= MAX_RUNTIME_SAMPLES) {
                            nextIndex = 0;
                        }
                        sumSamples -= runTimes[nextIndex];
                        runTimes[nextIndex++] = seconds;
                        sumSamples += seconds;
                    }

                    if (nextIndex >= MAX_RUNTIME_SAMPLES) {
                        nextIndex = 0;
                    }
                }
            }
        } while (uaArchiveMoveNext(hArchive));
    }

    if (storedSamples >= MIN_REQUIRED_SAMPLES) {
        sortDoubles(runTimes, storedSamples);
        if (storedSamples & 1) {
            median = runTimes[storedSamples / 2];
        } else {
            median = 0.5 * (runTimes[storedSamples / 2 - 1] + runTimes[storedSamples / 2]);
        }
    } else if (storedSamples > 0) {
        median = sumSamples / (double)storedSamples;
    } else {
        median = DEFAULT_ETA_SECONDS;
    }

cleanup:
    if (hArchive) {
        if (archiveOpened) uaArchiveClose(hArchive);
        uaReleaseArchive(hArchive);
    }
    if (hConnect) {
        uaDisconnect(hConnect);
    }
    SetTagDouble(resultTag, median);
}

static void sortDoubles(double* values, int count)
{
    int i;
    for (i = 1; i < count; ++i) {
        double current = values[i];
        int j = i - 1;
        while (j >= 0 && values[j] > current) {
            values[j + 1] = values[j];
            --j;
        }
        values[j + 1] = current;
    }
}

static int parseRuntimeSeconds(const char* runtime, double* seconds)
{
    const char* p;
    const char* colon;
    long hours = 0;
    long minutes = 0;

    if (!runtime || !*runtime || !seconds) {
        return 0;
    }

    p = skipSpace(runtime);
    colon = p;
    while (*colon && *colon != ':') {
        ++colon;
    }

    if (*colon != ':') {
        return 0;
    }

    if (!parseNonNegativeLong(p, &colon, &hours)) {
        return 0;
    }

    colon = skipSpace(colon);
    if (*colon != ':') {
        return 0;
    }

    p = skipSpace(colon + 1);
    if (!parseNonNegativeLong(p, &colon, &minutes)) {
        return 0;
    }

    colon = skipSpace(colon);
    if (*colon != '\0' || minutes < 0 || minutes >= 60 || hours < 0) {
        return 0;
    }

    *seconds = (double)hours * 3600.0 + (double)minutes * 60.0;
    return 1;
}

static int parseNonNegativeLong(const char* text, const char** endOut, long* valueOut)
{
    const char* p = text;
    long value = 0;
    int digits = 0;

    if (!text || !valueOut) {
        return 0;
    }

    while (*p >= '0' && *p <= '9') {
        long nextValue = value * 10 + (*p - '0');
        if (nextValue < value) {
            return 0;
        }
        value = nextValue;
        ++p;
        ++digits;
    }

    if (digits == 0) {
        return 0;
    }

    if (endOut) {
        *endOut = p;
    }

    *valueOut = value;
    return 1;
}

static const char* skipSpace(const char* text)
{
    const char* p = text;
    while (p && *p && isSpaceChar(*p)) {
        ++p;
    }
    return p ? p : text;
}

static int isSpaceChar(char c)
{
    return c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\f' || c == '\v';
}

static unsigned int textLength(const char* text)
{
    unsigned int len = 0;
    if (!text) {
        return 0;
    }

    while (text[len] != '\0') {
        ++len;
    }

    return len;
}

static int textCompare(const char* a, const char* b)
{
    unsigned char ca;
    unsigned char cb;

    if (!a && !b) return 0;
    if (!a) return -1;
    if (!b) return 1;

    while (*a || *b) {
        ca = (unsigned char)*a;
        cb = (unsigned char)*b;
        if (ca != cb) {
            return (ca < cb) ? -1 : 1;
        }
        if (*a == '\0') {
            break;
        }
        ++a;
        ++b;
    }

    return 0;
}

static void copyTrim(char* dest, unsigned int destSize, const char* src)
{
    const char* start;
    const char* end;
    unsigned int len;
    unsigned int i;

    if (!dest || destSize == 0) {
        return;
    }

    dest[0] = '\0';
    if (!src) {
        return;
    }

    start = src;
    while (*start && isSpaceChar(*start)) {
        ++start;
    }

    end = start + textLength(start);
    while (end > start && isSpaceChar(*(end - 1))) {
        --end;
    }

    len = (unsigned int)(end - start);
    if (len >= destSize) {
        len = destSize - 1;
    }

    for (i = 0; i < len; ++i) {
        dest[i] = start[i];
    }
    dest[len] = '\0';
}
