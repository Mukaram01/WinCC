// UpdateFormattedTime.c — HH:MM from numeric seconds
#include "apdefap.h"

void UpdateFormattedTime(const char* sourceTag, const char* targetTag)
{
    double secs;
    int total;
    int hrs;
    int mins;
    char buf[16];

    if (!sourceTag || sourceTag[0] == '\0' || !targetTag || targetTag[0] == '\0') return;

    secs = GetTagDouble(sourceTag);
    if (secs < 0.0) secs = 0.0;

    total = (int)secs;
    /* Wrap hours at 24 so this tag presents a clock-style time rather than accumulated hours */
    hrs   = (total / 3600) % 24;
    mins  = (total % 3600) / 60;

    sprintf(buf, "%02d:%02d", hrs, mins);
    SetTagChar(targetTag, buf);
}
