/* WriteValuesToUserArchive_V2.c — ProductionEvents writer (human-first)
   Archive field order (1-based):
     1  MachineName       (String)
     2  OrderNumber       (String)
     3  SerialNumber      (String)
     4  OperationNumber   (String)
     5  PartNumber        (String)
     6  RuntimeHHMM       (String, "HH:MM")
     7  DowntimeHHMM      (String, "HH:MM")
     8  StartDate         (String, "DD/MM/YYYY")
     9  StartTime         (String, "HH:MM")
     10 PlannedEndDate    (String, "DD/MM/YYYY")
     11 PlannedEndTime    (String, "HH:MM")
*/

#include "apdefap.h"
#include <string.h>

/* --- field indices for ProductionEvents --- */
#define F_MACHINE        1
#define F_ORDER          2
#define F_SERIAL         3
#define F_OPERATION      4
#define F_PART           5
#define F_RUNTIME_HHMM   6
#define F_DOWNTIME_HHMM  7
#define F_START_DATE     8
#define F_START_TIME     9
#define F_END_DATE       10
#define F_END_TIME       11

/* Safe fallback for NULL strings */
static const char* nz(const char* s) { return s ? s : ""; }

static int parse_hhmm_seconds(const char* text, double* secondsOut)
{
    const char* p;
    long hh = 0;
    int  mm  = 0;
    int  digits = 0;

    if (!text || !secondsOut) return 0;

    p = text;
    /* Parse hours: one or more digits */
    while (*p >= '0' && *p <= '9') {
        hh = hh * 10 + (*p - '0');
        ++p;
        ++digits;
    }
    if (digits == 0 || *p != ':') return 0;
    ++p; /* skip ':' */

    /* Parse minutes: exactly two digits */
    if (p[0] < '0' || p[0] > '9' || p[1] < '0' || p[1] > '9') return 0;
    mm = (p[0] - '0') * 10 + (p[1] - '0');
    p += 2;
    if (*p != '\0') return 0; /* reject trailing garbage */

    if (mm < 0 || mm > 59 || hh < 0) return 0;

    *secondsOut = (double)hh * 3600.0 + (double)mm * 60.0;
    return 1;
}

static void seconds_to_hhmm(double seconds, char out[16])
{
    long total;
    long hh;
    long mm;

    if (!out) return;
    if (seconds < 0.0) seconds = 0.0;

    total = (long)(seconds + 0.5);
    hh = total / 3600;
    mm = (total % 3600) / 60;

    sprintf(out, "%02ld:%02ld", hh, mm);
}

static int key_matches(const char* machineName,
                       const char* orderNumber,
                       const char* serialNumber,
                       const char* operationNumber,
                       const char* partNumber,
                       const char* startDate,
                       const char* recMachine,
                       const char* recOrder,
                       const char* recSerial,
                       const char* recOperation,
                       const char* recPart,
                       const char* recStartDate)
{
    return strcmp(nz(machineName),    nz(recMachine))    == 0 &&
           strcmp(nz(orderNumber),    nz(recOrder))      == 0 &&
           strcmp(nz(serialNumber),   nz(recSerial))     == 0 &&
           strcmp(nz(operationNumber),nz(recOperation))  == 0 &&
           strcmp(nz(partNumber),     nz(recPart))       == 0 &&
           strcmp(nz(startDate),      nz(recStartDate))  == 0;
}

void WriteValuesToUserArchive_V2(const char* archiveName,    /* "ProductionEvents" */
                                 const char* machineName,
                                 const char* orderNumber,
                                 const char* serialNumber,
                                 const char* operationNumber,
                                 const char* partNumber,
                                 const char* runtimeHHMM,     /* "HH:MM" */
                                 const char* downtimeHHMM,    /* "HH:MM" */
                                 const char* startDate,       /* "DD/MM/YYYY" */
                                 const char* startTime,       /* "HH:MM" */
                                 const char* plannedEndDate,  /* "DD/MM/YYYY" */
                                 const char* plannedEndTime)  /* "HH:MM" */
{
    UAHCONNECT hCon = 0;
    UAHARCHIVE hArc = 0;
    int ok;
    int writeOk = 0;
    int archiveOpened = 0;
    int hasExisting = 0;
    int duplicateCount = 0;

    char existingMachine[64];
    char existingOrder[64];
    char existingSerial[64];
    char existingOperation[64];
    char existingPart[64];
    char existingRun[16];
    char existingDown[16];
    char existingStartDate[16];
    char existingStartTime[16];
    char mergedRun[16];
    char mergedDown[16];
    double oldRunSeconds = 0.0;
    double oldDownSeconds = 0.0;
    double newRunSeconds = 0.0;
    double newDownSeconds = 0.0;

    if (!archiveName || archiveName[0] == '\0') {
        return;
    }

    /* Connect */
    ok = uaConnect(&hCon);
    if (!ok) return;

    /* Locate archive by name */
    ok = uaQueryArchiveByName(hCon, archiveName, &hArc);
    if (ok && hArc) {
        /* Open, set fields, insert */
        ok = uaArchiveOpen(hArc);
        if (ok) {
            archiveOpened = 1;

            /* Deterministic de-dupe key:
               MachineName + OrderNumber + SerialNumber + OperationNumber + PartNumber + StartDate
               StartDate is included so that the same job on a different production day
               always creates a new row rather than updating a historical entry. */
            if (uaArchiveMoveFirst(hArc)) {
                do {
                    existingMachine[0]   = '\0';
                    existingOrder[0]     = '\0';
                    existingSerial[0]    = '\0';
                    existingOperation[0] = '\0';
                    existingPart[0]      = '\0';
                    existingStartDate[0] = '\0';

                    uaArchiveGetFieldValueString(hArc, F_MACHINE,    existingMachine,    sizeof(existingMachine));
                    uaArchiveGetFieldValueString(hArc, F_ORDER,      existingOrder,      sizeof(existingOrder));
                    uaArchiveGetFieldValueString(hArc, F_SERIAL,     existingSerial,     sizeof(existingSerial));
                    uaArchiveGetFieldValueString(hArc, F_OPERATION,  existingOperation,  sizeof(existingOperation));
                    uaArchiveGetFieldValueString(hArc, F_PART,       existingPart,       sizeof(existingPart));
                    uaArchiveGetFieldValueString(hArc, F_START_DATE, existingStartDate,  sizeof(existingStartDate));

                    if (key_matches(machineName, orderNumber, serialNumber, operationNumber, partNumber, startDate,
                                    existingMachine, existingOrder, existingSerial, existingOperation, existingPart, existingStartDate)) {
                        hasExisting = 1;

                        existingRun[0]       = '\0';
                        existingDown[0]      = '\0';
                        existingStartTime[0] = '\0';

                        uaArchiveGetFieldValueString(hArc, F_RUNTIME_HHMM,  existingRun,       sizeof(existingRun));
                        uaArchiveGetFieldValueString(hArc, F_DOWNTIME_HHMM, existingDown,      sizeof(existingDown));
                        uaArchiveGetFieldValueString(hArc, F_START_TIME,    existingStartTime, sizeof(existingStartTime));

                        if (!parse_hhmm_seconds(existingRun, &oldRunSeconds)) oldRunSeconds = 0.0;
                        if (!parse_hhmm_seconds(existingDown, &oldDownSeconds)) oldDownSeconds = 0.0;
                        if (!parse_hhmm_seconds(nz(runtimeHHMM), &newRunSeconds)) newRunSeconds = 0.0;
                        if (!parse_hhmm_seconds(nz(downtimeHHMM), &newDownSeconds)) newDownSeconds = 0.0;

                        seconds_to_hhmm(oldRunSeconds + newRunSeconds, mergedRun);
                        seconds_to_hhmm(oldDownSeconds + newDownSeconds, mergedDown);

                        uaArchiveSetFieldValueString(hArc, F_MACHINE,       nz(machineName));
                        uaArchiveSetFieldValueString(hArc, F_ORDER,         nz(orderNumber));
                        uaArchiveSetFieldValueString(hArc, F_SERIAL,        nz(serialNumber));
                        uaArchiveSetFieldValueString(hArc, F_OPERATION,     nz(operationNumber));
                        uaArchiveSetFieldValueString(hArc, F_PART,          nz(partNumber));
                        uaArchiveSetFieldValueString(hArc, F_RUNTIME_HHMM,  mergedRun);
                        uaArchiveSetFieldValueString(hArc, F_DOWNTIME_HHMM, mergedDown);
                        /* Keep first value start date/time from existing row */
                        uaArchiveSetFieldValueString(hArc, F_START_DATE,    nz(existingStartDate[0] ? existingStartDate : startDate));
                        uaArchiveSetFieldValueString(hArc, F_START_TIME,    nz(existingStartTime[0] ? existingStartTime : startTime));
                        /* Keep second value planned end date/time from incoming row */
                        uaArchiveSetFieldValueString(hArc, F_END_DATE,      nz(plannedEndDate));
                        uaArchiveSetFieldValueString(hArc, F_END_TIME,      nz(plannedEndTime));

                        writeOk = uaArchiveUpdate(hArc);
                        break;
                    }
                } while (uaArchiveMoveNext(hArc));
            }

            if (!hasExisting) {
                uaArchiveSetFieldValueString(hArc, F_MACHINE,       nz(machineName));
                uaArchiveSetFieldValueString(hArc, F_ORDER,         nz(orderNumber));
                uaArchiveSetFieldValueString(hArc, F_SERIAL,        nz(serialNumber));
                uaArchiveSetFieldValueString(hArc, F_OPERATION,     nz(operationNumber));
                uaArchiveSetFieldValueString(hArc, F_PART,          nz(partNumber));
                uaArchiveSetFieldValueString(hArc, F_RUNTIME_HHMM,  nz(runtimeHHMM));
                uaArchiveSetFieldValueString(hArc, F_DOWNTIME_HHMM, nz(downtimeHHMM));
                uaArchiveSetFieldValueString(hArc, F_START_DATE,    nz(startDate));
                uaArchiveSetFieldValueString(hArc, F_START_TIME,    nz(startTime));
                uaArchiveSetFieldValueString(hArc, F_END_DATE,      nz(plannedEndDate));
                uaArchiveSetFieldValueString(hArc, F_END_TIME,      nz(plannedEndTime));

                writeOk = uaArchiveInsert(hArc);
            }

            if (!writeOk) {
                SetTagWord("ArchiveInsertFailed", 1);
            }

            /* Post-load duplicate check query: fail batch when duplicates exist */
            if (uaArchiveMoveFirst(hArc)) {
                do {
                    char chkMachine[64]   = "";
                    char chkOrder[64]     = "";
                    char chkSerial[64]    = "";
                    char chkOperation[64] = "";
                    char chkPart[64]      = "";
                    char chkStartDate[16] = "";

                    uaArchiveGetFieldValueString(hArc, F_MACHINE,    chkMachine,    sizeof(chkMachine));
                    uaArchiveGetFieldValueString(hArc, F_ORDER,      chkOrder,      sizeof(chkOrder));
                    uaArchiveGetFieldValueString(hArc, F_SERIAL,     chkSerial,     sizeof(chkSerial));
                    uaArchiveGetFieldValueString(hArc, F_OPERATION,  chkOperation,  sizeof(chkOperation));
                    uaArchiveGetFieldValueString(hArc, F_PART,       chkPart,       sizeof(chkPart));
                    uaArchiveGetFieldValueString(hArc, F_START_DATE, chkStartDate,  sizeof(chkStartDate));

                    if (key_matches(machineName, orderNumber, serialNumber, operationNumber, partNumber, startDate,
                                    chkMachine, chkOrder, chkSerial, chkOperation, chkPart, chkStartDate)) {
                        ++duplicateCount;
                    }
                } while (uaArchiveMoveNext(hArc));
            }

            if (duplicateCount > 1) {
                SetTagWord("ArchiveDuplicateCheckFailed", 1);
                SetTagWord("ArchiveInsertFailed", 1);
            }

            uaArchiveClose(hArc);
            archiveOpened = 0;
        }
        uaReleaseArchive(hArc);
    }

    if (hArc && archiveOpened) {
        uaArchiveClose(hArc);
    }
    uaDisconnect(hCon);
}
