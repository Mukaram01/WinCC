/* WriteValuesToUserArchive_V2.c — ProductionEvents writer (human-first)
   Archive field order (1-based):
     1  MachineName       (String)
     2  OrderNumber       (String)
     3  SerialNumber      (String)
     4  OperationNumber   (String)
     5  PartNumber        (String)
     6  RuntimeHHMM       (String, "HH:MM")
     7  DowntimeHHMM      (String, "HH:MM")
     8  StartDate         (String, "DD/MM/YYYY")
     9  StartTime         (String, "HH:MM")
     10 PlannedEndDate    (String, "DD/MM/YYYY")
     11 PlannedEndTime    (String, "HH:MM")
*/

#include "apdefap.h"

/* 5-minute threshold: extra archive rows with runtime >= this are merged;
   rows below this are discarded (too short to be meaningful). */
#define MIN_MERGE_RUNTIME_SECONDS (300.0)

/* Maximum number of extra duplicate row IDs collected per call. */
#define MAX_DUP_IDS 64

/* --- field indices for ProductionEvents --- */
#define F_MACHINE        1
#define F_ORDER          2
#define F_SERIAL         3
#define F_OPERATION      4
#define F_PART           5
#define F_RUNTIME_HHMM   6
#define F_DOWNTIME_HHMM  7
#define F_START_DATE     8
#define F_START_TIME     9
#define F_END_DATE       10
#define F_END_TIME       11

/* Safe fallback for NULL strings */
static const char* nz(const char* s) { return s ? s : ""; }

static int parse_hhmm_seconds(const char* text, double* secondsOut)
{
    const char* p;
    long hh = 0;
    int  mm  = 0;
    int  digits = 0;

    if (!text || !secondsOut) return 0;

    p = text;
    /* Parse hours: one or more digits */
    while (*p >= '0' && *p <= '9') {
        hh = hh * 10 + (*p - '0');
        ++p;
        ++digits;
    }
    if (digits == 0 || *p != ':') return 0;
    ++p; /* skip ':' */

    /* Parse minutes: exactly two digits */
    if (p[0] < '0' || p[0] > '9' || p[1] < '0' || p[1] > '9') return 0;
    mm = (p[0] - '0') * 10 + (p[1] - '0');
    p += 2;
    if (*p != '\0') return 0; /* reject trailing garbage */

    if (mm < 0 || mm > 59 || hh < 0) return 0;

    *secondsOut = (double)hh * 3600.0 + (double)mm * 60.0;
    return 1;
}

static void seconds_to_hhmm(double seconds, char out[16])
{
    long total;
    long hh;
    long mm;

    if (!out) return;
    if (seconds < 0.0) seconds = 0.0;

    total = (long)(seconds + 0.5);
    hh = total / 3600;
    mm = (total % 3600) / 60;

    sprintf(out, "%02ld:%02ld", hh, mm);
}

static int key_matches(const char* machineName,
                       const char* orderNumber,
                       const char* serialNumber,
                       const char* operationNumber,
                       const char* partNumber,
                       const char* startDate,
                       const char* recMachine,
                       const char* recOrder,
                       const char* recSerial,
                       const char* recOperation,
                       const char* recPart,
                       const char* recStartDate)
{
    return strcmp(nz(machineName),    nz(recMachine))    == 0 &&
           strcmp(nz(orderNumber),    nz(recOrder))      == 0 &&
           strcmp(nz(serialNumber),   nz(recSerial))     == 0 &&
           strcmp(nz(operationNumber),nz(recOperation))  == 0 &&
           strcmp(nz(partNumber),     nz(recPart))       == 0 &&
           strcmp(nz(startDate),      nz(recStartDate))  == 0;
}

void WriteValuesToUserArchive_V2(const char* archiveName,    /* "ProductionEvents" */
                                 const char* machineName,
                                 const char* orderNumber,
                                 const char* serialNumber,
                                 const char* operationNumber,
                                 const char* partNumber,
                                 const char* runtimeHHMM,     /* "HH:MM" */
                                 const char* downtimeHHMM,    /* "HH:MM" */
                                 const char* startDate,       /* "DD/MM/YYYY" */
                                 const char* startTime,       /* "HH:MM" */
                                 const char* plannedEndDate,  /* "DD/MM/YYYY" */
                                 const char* plannedEndTime)  /* "HH:MM" */
{
    UAHCONNECT hCon = 0;
    UAHARCHIVE hArc = 0;
    int ok;
    int writeOk = 0;
    int hasExisting = 0;

    char existingMachine[64];
    char existingOrder[64];
    char existingSerial[64];
    char existingOperation[64];
    char existingPart[64];
    char existingRun[16];
    char existingDown[16];
    char existingStartDate[16];
    char existingStartTime[16];
    char mergedRun[16];
    char mergedDown[16];
    double oldRunSeconds = 0.0;
    double oldDownSeconds = 0.0;
    double newRunSeconds = 0.0;
    double newDownSeconds = 0.0;

    if (!archiveName || archiveName[0] == '\0') {
        printf("WriteValuesToUserArchive_V2: archiveName is null/empty, aborting\n");
        return;
    }

    printf("WriteValuesToUserArchive_V2: entry archive=%s machine=%s order=%s serial=%s op=%s part=%s rt=%s down=%s\n",
           archiveName, nz(machineName), nz(orderNumber), nz(serialNumber),
           nz(operationNumber), nz(partNumber), nz(runtimeHHMM), nz(downtimeHHMM));
    printf("WriteValuesToUserArchive_V2: startDate=%s startTime=%s endDate=%s endTime=%s\n",
           nz(startDate), nz(startTime), nz(plannedEndDate), nz(plannedEndTime));

    /* Connect */
    ok = uaConnect(&hCon);
    if (!ok) {
        printf("WriteValuesToUserArchive_V2: uaConnect FAILED\n");
        SetTagWord("ArchiveInsertFailed", 1);
        return;
    }

    /* Locate archive by name */
    ok = uaQueryArchiveByName(hCon, archiveName, &hArc);
    if (!ok || !hArc) {
        printf("WriteValuesToUserArchive_V2: uaQueryArchiveByName FAILED for '%s'\n", archiveName);
        SetTagWord("ArchiveInsertFailed", 1);
        uaDisconnect(hCon);
        return;
    }
    /* Open, set fields, insert */
    ok = uaArchiveOpen(hArc);
    if (!ok) {
        printf("WriteValuesToUserArchive_V2: uaArchiveOpen FAILED for '%s'\n", archiveName);
        SetTagWord("ArchiveInsertFailed", 1);
        uaReleaseArchive(hArc);
        uaDisconnect(hCon);
        return;
    }
    {
        /* Deterministic de-dupe key:
           MachineName + OrderNumber + SerialNumber + OperationNumber + PartNumber + StartDate
           StartDate is included so that the same job on a different production day
           always creates a new row rather than updating a historical entry. */
        if (uaArchiveMoveFirst(hArc)) {
            do {
                existingMachine[0]   = '\0';
                existingOrder[0]     = '\0';
                existingSerial[0]    = '\0';
                existingOperation[0] = '\0';
                existingPart[0]      = '\0';
                existingStartDate[0] = '\0';

                uaArchiveGetFieldValueString(hArc, F_MACHINE,    existingMachine,    sizeof(existingMachine));
                uaArchiveGetFieldValueString(hArc, F_ORDER,      existingOrder,      sizeof(existingOrder));
                uaArchiveGetFieldValueString(hArc, F_SERIAL,     existingSerial,     sizeof(existingSerial));
                uaArchiveGetFieldValueString(hArc, F_OPERATION,  existingOperation,  sizeof(existingOperation));
                uaArchiveGetFieldValueString(hArc, F_PART,       existingPart,       sizeof(existingPart));
                uaArchiveGetFieldValueString(hArc, F_START_DATE, existingStartDate,  sizeof(existingStartDate));

                if (key_matches(machineName, orderNumber, serialNumber, operationNumber, partNumber, startDate,
                                existingMachine, existingOrder, existingSerial, existingOperation, existingPart, existingStartDate)) {
                    hasExisting = 1;

                    existingRun[0]       = '\0';
                    existingDown[0]      = '\0';
                    existingStartTime[0] = '\0';

                    uaArchiveGetFieldValueString(hArc, F_RUNTIME_HHMM,  existingRun,       sizeof(existingRun));
                    uaArchiveGetFieldValueString(hArc, F_DOWNTIME_HHMM, existingDown,      sizeof(existingDown));
                    uaArchiveGetFieldValueString(hArc, F_START_TIME,    existingStartTime, sizeof(existingStartTime));

                    if (!parse_hhmm_seconds(existingRun, &oldRunSeconds)) oldRunSeconds = 0.0;
                    if (!parse_hhmm_seconds(existingDown, &oldDownSeconds)) oldDownSeconds = 0.0;
                    if (!parse_hhmm_seconds(nz(runtimeHHMM), &newRunSeconds)) newRunSeconds = 0.0;
                    if (!parse_hhmm_seconds(nz(downtimeHHMM), &newDownSeconds)) newDownSeconds = 0.0;

                    seconds_to_hhmm(oldRunSeconds + newRunSeconds, mergedRun);
                    seconds_to_hhmm(oldDownSeconds + newDownSeconds, mergedDown);

                    printf("WriteValuesToUserArchive_V2: found existing row - updating\n");
                    printf("  existingRun=%s existingDown=%s -> mergedRun=%s mergedDown=%s\n",
                           existingRun, existingDown, mergedRun, mergedDown);

                    uaArchiveSetFieldValueString(hArc, F_MACHINE,       nz(machineName));
                    uaArchiveSetFieldValueString(hArc, F_ORDER,         nz(orderNumber));
                    uaArchiveSetFieldValueString(hArc, F_SERIAL,        nz(serialNumber));
                    uaArchiveSetFieldValueString(hArc, F_OPERATION,     nz(operationNumber));
                    uaArchiveSetFieldValueString(hArc, F_PART,          nz(partNumber));
                    uaArchiveSetFieldValueString(hArc, F_RUNTIME_HHMM,  mergedRun);
                    uaArchiveSetFieldValueString(hArc, F_DOWNTIME_HHMM, mergedDown);
                    /* Keep first value start date/time from existing row */
                    uaArchiveSetFieldValueString(hArc, F_START_DATE,    nz(existingStartDate[0] ? existingStartDate : startDate));
                    uaArchiveSetFieldValueString(hArc, F_START_TIME,    nz(existingStartTime[0] ? existingStartTime : startTime));
                    /* Keep second value planned end date/time from incoming row */
                    uaArchiveSetFieldValueString(hArc, F_END_DATE,      nz(plannedEndDate));
                    uaArchiveSetFieldValueString(hArc, F_END_TIME,      nz(plannedEndTime));

                    writeOk = uaArchiveUpdate(hArc);
                    printf("WriteValuesToUserArchive_V2: uaArchiveUpdate result=%d\n", writeOk);
                    break;
                }
            } while (uaArchiveMoveNext(hArc));
        }

        if (!hasExisting) {
            printf("WriteValuesToUserArchive_V2: no existing row found - inserting new row\n");
            uaArchiveSetFieldValueString(hArc, F_MACHINE,       nz(machineName));
            uaArchiveSetFieldValueString(hArc, F_ORDER,         nz(orderNumber));
            uaArchiveSetFieldValueString(hArc, F_SERIAL,        nz(serialNumber));
            uaArchiveSetFieldValueString(hArc, F_OPERATION,     nz(operationNumber));
            uaArchiveSetFieldValueString(hArc, F_PART,          nz(partNumber));
            uaArchiveSetFieldValueString(hArc, F_RUNTIME_HHMM,  nz(runtimeHHMM));
            uaArchiveSetFieldValueString(hArc, F_DOWNTIME_HHMM, nz(downtimeHHMM));
            uaArchiveSetFieldValueString(hArc, F_START_DATE,    nz(startDate));
            uaArchiveSetFieldValueString(hArc, F_START_TIME,    nz(startTime));
            uaArchiveSetFieldValueString(hArc, F_END_DATE,      nz(plannedEndDate));
            uaArchiveSetFieldValueString(hArc, F_END_TIME,      nz(plannedEndTime));

            writeOk = uaArchiveInsert(hArc);
            printf("WriteValuesToUserArchive_V2: uaArchiveInsert result=%d\n", writeOk);
        }

        if (!writeOk) {
            printf("WriteValuesToUserArchive_V2: write FAILED (hasExisting=%d), setting ArchiveInsertFailed\n",
                   hasExisting);
            SetTagWord("ArchiveInsertFailed", 1);
        }

        /* Post-write deduplication with merge-or-discard for extra rows.
           Rules (applied after every write):
             - The survivor is the first matching row (already updated with the
               incoming data in the write phase above: merged runtime/downtime,
               preserved start date/time, updated planned end date/time).
             - Extra rows (2nd, 3rd, … match) with RuntimeHHMM >= 5 min (300 s)
               are MERGED: their runtime and downtime are folded into the survivor,
               then the extra row is deleted.
             - Extra rows with RuntimeHHMM < 5 min are DELETED without merging.
           Start date/time  : survivor's value is used (kept from the write phase).
           Planned end date/time: incoming value is used (set in the write phase). */
        {
            long   survivorID    = -1L;
            double extraRunSecs  = 0.0;
            double extraDownSecs = 0.0;
            int    extraMerged   = 0;
            int    deleteCount   = 0;
            int    deleteIDCount = 0;
            long   deleteIDs[MAX_DUP_IDS];
            int    i;

            /* Pass 1 — identify the survivor and collect extra rows */
            if (uaArchiveMoveFirst(hArc)) {
                do {
                    char chkMachine[64]   = "";
                    char chkOrder[64]     = "";
                    char chkSerial[64]    = "";
                    char chkOperation[64] = "";
                    char chkPart[64]      = "";
                    char chkStartDate[16] = "";
                    long rowID = -1L;

                    uaArchiveGetFieldValueString(hArc, F_MACHINE,    chkMachine,    sizeof(chkMachine));
                    uaArchiveGetFieldValueString(hArc, F_ORDER,      chkOrder,      sizeof(chkOrder));
                    uaArchiveGetFieldValueString(hArc, F_SERIAL,     chkSerial,     sizeof(chkSerial));
                    uaArchiveGetFieldValueString(hArc, F_OPERATION,  chkOperation,  sizeof(chkOperation));
                    uaArchiveGetFieldValueString(hArc, F_PART,       chkPart,       sizeof(chkPart));
                    uaArchiveGetFieldValueString(hArc, F_START_DATE, chkStartDate,  sizeof(chkStartDate));

                    if (key_matches(machineName, orderNumber, serialNumber, operationNumber, partNumber, startDate,
                                    chkMachine, chkOrder, chkSerial, chkOperation, chkPart, chkStartDate)) {
                        if (!uaArchiveGetFieldValueLong(hArc, 0, &rowID)) rowID = -1L;

                        if (survivorID == -1L) {
                            /* First match = survivor (already updated above) */
                            survivorID = rowID;
                        } else {
                            /* Extra row: merge if runtime >= 5 min, else discard */
                            char   extraRun[16]  = "";
                            char   extraDown[16] = "";
                            double rowRunSecs    = 0.0;
                            double rowDownSecs   = 0.0;

                            uaArchiveGetFieldValueString(hArc, F_RUNTIME_HHMM,  extraRun,  sizeof(extraRun));
                            uaArchiveGetFieldValueString(hArc, F_DOWNTIME_HHMM, extraDown, sizeof(extraDown));
                            if (!parse_hhmm_seconds(extraRun,  &rowRunSecs))  rowRunSecs  = 0.0;
                            if (!parse_hhmm_seconds(extraDown, &rowDownSecs)) rowDownSecs = 0.0;

                            if (rowRunSecs >= MIN_MERGE_RUNTIME_SECONDS) {
                                extraRunSecs  += rowRunSecs;
                                extraDownSecs += rowDownSecs;
                                ++extraMerged;
                                printf("WriteValuesToUserArchive_V2: queuing merge of extra row ID=%ld rt=%s down=%s\n",
                                       rowID, extraRun, extraDown);
                            } else {
                                printf("WriteValuesToUserArchive_V2: discarding short extra row ID=%ld rt=%s (< 5 min)\n",
                                       rowID, extraRun);
                            }

                            if (deleteIDCount < MAX_DUP_IDS) {
                                deleteIDs[deleteIDCount++] = rowID;
                            } else {
                                printf("WriteValuesToUserArchive_V2: MAX_DUP_IDS limit reached, extra row ID=%ld not queued for deletion\n",
                                       rowID);
                                SetTagWord("ArchiveDuplicateCheckFailed", 1);
                            }
                        }
                    }
                } while (uaArchiveMoveNext(hArc));
            }

            /* Pass 2 — if extra rows were merged, fold their totals into the
               survivor.  Start date/time and planned end are already correct
               on the survivor from the write phase; only runtime/downtime change. */
            if (extraMerged > 0 && survivorID >= 0L) {
                if (uaArchiveMoveFirst(hArc)) {
                    do {
                        long rowID = -1L;
                        if (uaArchiveGetFieldValueLong(hArc, 0, &rowID) && rowID == survivorID) {
                            char   survRun[16]   = "";
                            char   survDown[16]  = "";
                            char   newRun[16]    = "";
                            char   newDown[16]   = "";
                            double survRunSecs   = 0.0;
                            double survDownSecs  = 0.0;

                            uaArchiveGetFieldValueString(hArc, F_RUNTIME_HHMM,  survRun,  sizeof(survRun));
                            uaArchiveGetFieldValueString(hArc, F_DOWNTIME_HHMM, survDown, sizeof(survDown));
                            if (!parse_hhmm_seconds(survRun,  &survRunSecs))  survRunSecs  = 0.0;
                            if (!parse_hhmm_seconds(survDown, &survDownSecs)) survDownSecs = 0.0;

                            seconds_to_hhmm(survRunSecs  + extraRunSecs,  newRun);
                            seconds_to_hhmm(survDownSecs + extraDownSecs, newDown);

                            printf("WriteValuesToUserArchive_V2: survivor ID=%ld adding %.0f s extra rt: %s->%s down: %s->%s\n",
                                   survivorID, extraRunSecs, survRun, newRun, survDown, newDown);

                            uaArchiveSetFieldValueString(hArc, F_RUNTIME_HHMM,  newRun);
                            uaArchiveSetFieldValueString(hArc, F_DOWNTIME_HHMM, newDown);
                            if (!uaArchiveUpdate(hArc)) {
                                printf("WriteValuesToUserArchive_V2: survivor merge update FAILED err=%d\n",
                                       uaGetLastError());
                            }
                            break;
                        }
                    } while (uaArchiveMoveNext(hArc));
                }
            }

            /* Pass 3 — delete every collected extra row by primary key.
               deleteIDs[i] may be -1L if uaArchiveGetFieldValueLong failed for
               that row; skip those to avoid constructing a malformed WHERE clause. */
            for (i = 0; i < deleteIDCount; ++i) {
                char whereClause[32] = "";
                if (deleteIDs[i] >= 0L) {
                    sprintf(whereClause, "ID=%ld", deleteIDs[i]);
                    if (uaArchiveDelete(hArc, whereClause)) {
                        ++deleteCount;
                        printf("WriteValuesToUserArchive_V2: deleted extra row ID=%ld\n", deleteIDs[i]);
                    } else {
                        printf("WriteValuesToUserArchive_V2: delete extra row FAILED ID=%ld err=%d\n",
                               deleteIDs[i], uaGetLastError());
                    }
                }
            }

            if (deleteCount > 0 || extraMerged > 0) {
                printf("WriteValuesToUserArchive_V2: dedup done merged=%d deleted=%d machine=%s order=%s\n",
                       extraMerged, deleteCount, nz(machineName), nz(orderNumber));
            }
            /* Flag residual extras that could not be deleted */
            if (deleteIDCount > 0 && deleteCount < deleteIDCount) {
                printf("WriteValuesToUserArchive_V2: dedup issue deleteIDCount=%d deleteCount=%d\n",
                       deleteIDCount, deleteCount);
                SetTagWord("ArchiveDuplicateCheckFailed", 1);
                SetTagWord("ArchiveInsertFailed", 1);
            }
        }

        uaArchiveClose(hArc);
    }
    uaReleaseArchive(hArc);
    uaDisconnect(hCon);
}
